\subsection{Детали реализации механизма в исходном коде}

\subsubsection{Виртуальный уровень и политики}

Одной из ключевых структур данных подсистемы управления памятью Linux является \texttt{vm\_area\_struct}. Она описывает диапазон пользовательской памяти, содержит информацию о его свойствах и семантике. Практически любая операция управления памятью в ядре начинается с обращения к соответствующему экземпляру \texttt{vm\_area\_struct}.

Одним из важнейших атрибутов данной структуры являются флаги (см листинг~\ref{lst:vm_area_struct}). Они представляют собой битовую маску, в которой каждый бит соответствует определённому свойству области памяти.

\begin{lstlisting}[
    style=normal,
    caption={Флаги виртуальной области памяти},
    label={lst:vm_area_struct}
]
/* include/linux/mm_types.h */

struct vm_area_struct {
    ...
	union {
		const vm_flags_t vm_flags;
		vm_flags_t __private __vm_flags;
	};
    ...
};
\end{lstlisting}

Для поддержки механизма репликации были добавлены два новых флага, отвечающих за репликацию таблиц трансляций и данных соответственно (см листинг~\ref{lst:vm_replica_flags}). Именно по ним и будет определяться поведение механизма репликации в отношении конкретной области памяти.

Однако сами флаги репликации устанавливаются в соответствии с выбранными политиками репликации, на которых основан пользовательский интерфейс механизма (см главу~\ref{sec:Chapter3_user_API}). Политики задаются на уровне виртуального адресного пространства процесса, которое в Linux представляется структурой \texttt{mm\_struct}. Для хранения текущих настроек репликации в неё было добавлено соответствующее поле (см листинг~\ref{lst:mm_struct}).

Таким образом, реализована инфраструктура, необходимая для назначения виртуальным областям памяти соответствующих флагов репликации. Установка флагов должна выполняться всякий раз, когда создаётся новая виртуальная область либо изменяются атрибуты уже существующей области, например, при обработке системных вызовов \texttt{mmap} и \texttt{mprotect}. Соответствующие изменения показаны в листинге~\ref{lst:vmflags_modify_replica}.

	return (vm_flags & VM_REPLICA_INIT) && (vm_flags & VM_ACCESS_FLAGS) &&

/* set replication flags of @vm_flags according to @mm's policies */
static inline void vmflags_modify_replica(
    struct mm_struct *mm, unsigned long *vm_flags) {
	*vm_flags &= ~(VM_REPLICA_INIT | VM_REPLICA_COMMIT);

		(get_table_replication_policy(mm) == TABLE_REPLICATION_MINIMAL &&
			__vmflags_replica_candidate(*vm_flags))) {

				vmflags_might_be_replicated(*vm_flags))
			*vm_flags |= VM_REPLICA_COMMIT;
	}
}

Помимо этого флаги репликации приходится синхронно обновлять существующим областям при изменении процессом политик репликации таблиц трансляций и/или данных. На листинге~\ref{lst:numa_mm_handle_table_replication} показан код для смен политик репликации таблиц с \texttt{All} на \texttt{Minimal} и с \texttt{None} на \texttt{Minimal}. Кроме обновления флагов подходящим виртуальным областям инициируются операции репликации и дерепликации таблиц трансляций. Код для остальных переходов между политиками устроен аналогичным образом и отличается лишь набором областей, состояние которых необходимо изменить.

/* mm/numa_user_replication.c */

/* called when table replication policy goes from All to Minimal */
static void dereplicate_all2min_pgtables(struct mm_struct *mm) {
	struct vm_area_struct *vma;
    ...
	do {
		unsigned long addr = vma->vm_start;
		struct vm_area_struct *next;
        ...
        /* clear VM_REPLICA_INIT and dereplicate
         * pgtables for not replica candidate vmas */
		if (numa_is_vma_replicant(vma) && !vma_might_be_replicated(vma)) {
			vm_flags_clear(vma, VM_REPLICA_INIT);
			...
			dereplicate_pgtables_pgd_range(&tlb, addr, vma->vm_end, ...);
		}
		vma = next;
	} while (vma);
    ...
}

static void numa_mm_table_replication_minimal(struct mm_struct *mm) {
	switch (get_table_replication_policy(mm)) {
    /* executed when table replication policy goes from None to Minimal */
	case TABLE_REPLICATION_NONE: {
		struct vm_area_struct *vma;
		MA_STATE(mas, &mm->mm_mt, 0, 0);

		mas_for_each(&mas, vma, ULONG_MAX) {
            /* set VM_REPLICA_INIT and replicate
             * pgtables for replica candidate vmas */
			if (vma_replica_candidate(vma)) {
				vm_flags_set(vma, VM_REPLICA_INIT);
				replicate_pgtables_vma(vma);
			}
		}
		break;
	}
	case TABLE_REPLICATION_MINIMAL:
		return;
	case TABLE_REPLICATION_ALL:
		dereplicate_all2min_pgtables(mm);
		break;
	}

	set_table_replication_policy(mm, TABLE_REPLICATION_MINIMAL);
}
\end{lstlisting}

Аналогичным образом реализовано изменение политики репликации данных. На листинге~\ref{lst:numa_mm_handle_data_replication} показаны переходы с \texttt{Lazy}/\texttt{All} на \texttt{None} и с \texttt{Lazy/None} на \texttt{All}. Вместе со сменой флагов выполняется репликация и дерепликация соответствующих физических страниц. Однако следует отметить, что при переходе с \texttt{None} на \texttt{Lazy} выполняется только обновление флагов, репликация данных осуществляется отложено при обработках \textit{NUMA hinting faults} (см главу~\ref{sec:Chapter3_data_repl_policies}).

\begin{lstlisting}[
    style=normal,
    caption={Установка флага репликации данных при смене политики},
    label={lst:numa_mm_handle_data_replication}
]
/* mm/numa_user_replication.c */

static bool numa_mm_data_replication_none(struct mm_struct *mm) {
	switch (get_data_replication_policy(mm)) {
	case DATA_REPLICATION_NONE:
		return true;
    /* executed when data replication policy
     * goes from Lazy/All to None */
	case DATA_REPLICATION_ON_DEMAND:
	case DATA_REPLICATION_ALL: {
		struct vm_area_struct *vma;
		MA_STATE(mas, &mm->mm_mt, 0, 0);

		mas_for_each(&mas, vma, ULONG_MAX) {
            /* deduplicate data and clear VM_REPLICA_COMMIT
             * for all has_replicas vmas */
			if (vma_has_replicas(vma)) {
				vma_start_write(vma);
				if (phys_deduplicate(vma,
						vma->vm_start, vma->vm_end - vma->vm_start))
					return false;
				__vm_flags_mod(vma, VM_NONE, VM_REPLICA_COMMIT);
			}
		}
		break;
	}
	}

	set_data_replication_policy(mm, DATA_REPLICATION_NONE);
	return true;
}

static bool numa_mm_data_replication_all(struct mm_struct *mm) {
	switch (get_data_replication_policy(mm)) {
    /* executed when data replication policy
     * goes from Lazy/None to All */
	case DATA_REPLICATION_NONE:
	case DATA_REPLICATION_ON_DEMAND: {
		struct vm_area_struct *vma;
		MA_STATE(mas, &mm->mm_mt, 0, 0);

		mas_for_each(&mas, vma, ULONG_MAX) {
            /* set VM_REPLICA_COMMIT an duplicate data
             * for all replica candidate vmas */
			if (vma_might_be_replicated(vma))
				vm_flags_set(vma, VM_REPLICA_COMMIT);
			if (vma_has_replicas(vma))
				phys_duplicate(vma,
                    vma->vm_start, vma->vm_end - vma->vm_start);
		}
		break;
	}
	case DATA_REPLICATION_ALL:
		return true;
	}

	set_data_replication_policy(mm, DATA);
	return true;
}
\end{lstlisting}
